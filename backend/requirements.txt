# bcrypt is pinned below 4.1 because passlib 1.7.4 reads bcrypt.__about__,
# which bcrypt>=4.1 removed -- newer bcrypt breaks passlib's hash/verify.

# JWT issuance/verification (Story 1.4, FR-1/FR-2). HS256 only needs
# stdlib hmac/hashlib -- no extra cryptography backend to pull in.

# Document parsing (Story 2.3): PDF text/outline extraction and HTML
# tag-stripping. beautifulsoup4 uses the stdlib `html.parser` backend at
# call sites -- no `lxml` dependency needed.
pypdf==6.15.0
beautifulsoup4==4.15.0

# DOCX/PPTX text extraction. python-docx pulls in lxml as a real
# dependency (unlike beautifulsoup4 above); python-pptx pulls in
# XlsxWriter and Pillow, neither used here -- both are the packages'
# own transitive requirements, not something this app calls directly.
python-docx==1.2.0
python-pptx==1.0.2

# No embedding library. Passage embeddings are computed by Weaviate
# itself (text2vec-weaviate), not in this process -- see
# shared/data_access/weaviate_client.py. `fastembed` was removed here:
# even as a quantized ONNX model, the multilingual MiniLM it ran needed
# ~554MB resident on top of a ~143MB app, against Render's 512MB free
# instance. Do not add sentence-transformers/torch either, for the same
# reason at a larger scale.
